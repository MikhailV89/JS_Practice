   indexOf([1, 23, 4, 1, 23, 4, 1, 23, 4], 4, 0) // вертає 2, бо починає шукать з 0го індекса
   indexOf([1, 23, 4, 1, 23, 4, 1, 23, 4], 4) // вертає 2, бо починає шукать з 0го індекса
